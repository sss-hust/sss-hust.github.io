第二节介绍的是并查集，可以很方便的查询两个节点是否在一起，分为如下阶段：

动态连接阶段为最初始阶段，代码如下，connected查询两个点是否连接，union使用轮询的办法将两个点连接起来，时间效率非常低,因为每一个点的孩子都会别变成新连接点的孩子。
```java
 public class QuickFindUF
 {
   private int[] id;
   public QuickFindUF(int N)
   {
      id = new int[N];
      for (int i = 0; i < N; i++)
         id[i] = i;
   }

   public boolean connected(int p, int q)
   {  return id[p] == id[q];  }

   public void union(int p, int q)
   {
      int pid = id[p];
      int qid = id[q];
      for (int i = 0; i < id.length; i++)
         if (id[i] == pid) id[i] = qid;
   }
 }
```

快速查找在原先的基础上，修改思路，union改为将自己以及孩子当成树连接在新的节点上，从而使得访问更快，代码如下：
```java
 public class QuickUnionUF
 {
   private int[] id;
   public QuickUnionUF(int N)
   {
      id = new int[N];
      for (int i = 0; i < N; i++) id[i] = i;
   }

   private int root(int i)
   {
      while (i != id[i]) i = id[i];
      return i;
   }

   public boolean connected(int p, int q)
   {
      return root(p) == root(q);
   }

   public void union(int p, int q)
   {
      int i = root(p);
      int j = root(q);
      id[i] = j;
   }
 }
```

最后的优化方向为合并时减少树的高度，首先两个树合并时，将小树合并在高树中，减少整体高度；其次，进行路径压缩，在查找时将每个查找的节点都移动到其父节点，依次递归到根节点，代码如下：
```java
 public int find(int p) {
        validate(p);
        while (p != parent[p]) {
            parent[p] = parent[parent[p]];    // path compression by halving
            p = parent[p];
        }
        return p;
    }

  public void union(int p, int q) {
        int rootP = find(p);
        int rootQ = find(q);
        if (rootP == rootQ) return;

        // make root of smaller rank point to root of larger rank
        if      (rank[rootP] < rank[rootQ]) parent[rootP] = rootQ;
        else if (rank[rootP] > rank[rootQ]) parent[rootQ] = rootP;
        else {
            parent[rootQ] = rootP;
            rank[rootP]++;
        }
        count--;
    }
```

章节测试需要完成两个类的设计，首先是渗流的模型建模，利用给出的uf模型可以很快完成；接着是完成数学计算模型，利用统计计算渗流阈值，反流问题采用两个uf模型，一个用于判断是否为反流，一个用于判断是否为渗流，代码如下：
```java
import edu.princeton.cs.algs4.WeightedQuickUnionUF;

public class Percolation {
    private int gridSize;
    private boolean[][] grid;
    private WeightedQuickUnionUF uf;
    private WeightedQuickUnionUF uffull;
    private int virtualTop;
    private int virtualBottom;
    private int numberOfOpenSites = 0;

    // creates n-by-n grid, with all sites initially blocked
    public Percolation(int n) {
        if (n <= 0) {
            throw new IllegalArgumentException("索引错误");
        }
        gridSize = n;
        grid = new boolean[n + 1][n + 1];
        uf = new WeightedQuickUnionUF(n * n + 2);
        uffull = new WeightedQuickUnionUF(n * n + 1);
        virtualTop = 0;
        virtualBottom = n * n + 1;
    }

    private void test(int x, int y) {
        if (x < 1 || x > gridSize || y < 1 || y > gridSize)
            throw new IllegalArgumentException("索引错误");

    }

    private int xyToID(int x, int y) {
        test(x, y);
        return (x - 1) * gridSize + y;
    }


    // opens the site (row, col) if it is not open already
    public void open(int row, int col) {
        test(row, col);
        if (isOpen(row, col)) return;
        if (row == 1) {
            uf.union(virtualTop, xyToID(row, col));
            uffull.union(virtualTop, xyToID(row, col));
        }
        if (row == gridSize) {
            uf.union(virtualBottom, xyToID(row, col));
        }
        grid[row][col] = true;
        numberOfOpenSites++;
        // 上下左右一一判断，只会通过扩展数组实现统一，待学习更好的方法
        if (row - 1 > 0 && isOpen(row - 1, col)) {
            uf.union(xyToID(row, col), xyToID(row - 1, col));
            uffull.union(xyToID(row, col), xyToID(row - 1, col));
        }

        if (row + 1 <= gridSize && isOpen(row + 1, col)) {
            uf.union(xyToID(row, col), xyToID(row + 1, col));
            uffull.union(xyToID(row, col), xyToID(row + 1, col));
        }

        if (col - 1 > 0 && isOpen(row, col - 1)) {
            uf.union(xyToID(row, col), xyToID(row, col - 1));
            uffull.union(xyToID(row, col), xyToID(row, col - 1));
        }

        if (col + 1 <= gridSize && isOpen(row, col + 1)) {
            uf.union(xyToID(row, col), xyToID(row, col + 1));
            uffull.union(xyToID(row, col), xyToID(row, col + 1));
        }

    }

    // is the site (row, col) open?
    public boolean isOpen(int row, int col) {
        test(row, col);
        return grid[row][col];
    }

    // is the site (row, col) full?
    public boolean isFull(int row, int col) {
        test(row, col);
        return isOpen(row, col) && uffull.find(virtualTop) == uffull.find(xyToID(row, col));
    }

    // returns the number of open sites
    public int numberOfOpenSites() {
        return numberOfOpenSites;
    }

    // does the system percolate?
    public boolean percolates() {
        return uf.find(virtualTop) == uf.find(virtualBottom);
    }

    // test client (optional)
    public static void main(String[] args) {
    }

}
```

import edu.princeton.cs.algs4.StdRandom;
import edu.princeton.cs.algs4.StdStats;

public class PercolationStats {
    private static final double CONFIDENCE = 1.96;
    private double[] thresholds;  // 存储每次实验的渗透阈值
    private int trials;           // 实验次数
    private double mean;         // 样本均值
    private double stddev;       // 样本标准差
    private double confidenceLo;
    private double confidenceHi;

    // perform independent trials on an n-by-n grid
    public PercolationStats(int n, int trials) {
        if (n <= 0 || trials <= 0)
            throw new IllegalArgumentException("索引错误");
        this.trials = trials;
        this.thresholds = new double[trials];
        for (int i = 0; i < trials; i++) {
            Percolation p = new Percolation(n);
            while (!p.percolates()) {
                int row = StdRandom.uniformInt(1, n + 1);
                int col = StdRandom.uniformInt(1, n + 1);
                if (p.isOpen(row, col)) continue;
                p.open(row, col);
            }
            thresholds[i] = 1.0 * p.numberOfOpenSites() / (n * n);
        }
        mean = StdStats.mean(thresholds);
        stddev = StdStats.stddev(thresholds);
        confidenceLo = mean - CONFIDENCE * stddev / Math.sqrt(trials);
        confidenceHi = mean + CONFIDENCE * stddev / Math.sqrt(trials);
    }

    public static void main(String[] args) {
        int n = Integer.parseInt(args[0]);
        int trial = Integer.parseInt(args[1]);
        if (n <= 0 || trial <= 0)
            throw new IllegalArgumentException("索引错误");
        PercolationStats stats = new PercolationStats(n, trial);
        System.out.println("mean = " + stats.mean);
        System.out.println("stddev = " + stats.stddev);
        System.out.println("95% confidence interval = [" + stats.confidenceLo +
                                   ", " + stats.confidenceHi + "]");
    }

    // sample mean of percolation threshold
    public double mean() {
        return mean;
    }

    // sample standard deviation of percolation threshold
    public double stddev() {
        if (trials == 1) {
            return Double.NaN;
        }
        return stddev;
    }

    // low endpoint of 95% confidence interval
    public double confidenceLo() {
        return confidenceLo;
    }

    // high endpoint of 95% confidence interval
    public double confidenceHi() {
        return confidenceHi;
    }
}
